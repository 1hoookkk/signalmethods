constexpr int kCols = 48, kRowsPlane = 24;
constexpr double kRootLo = 24.0, kRootHi = 120.0, kVoicingHi = 6.0;
const char* const kAxisLabel[3] = { "ROOT", "VOICING", "RESONANCE" };

double widthFromZ (double z) { return std::pow (13.0, 1.0 - z) - 1.0; }
}

juce::Rectangle<float> Workstation::planeRect() const
{
    return { L.field.getX() + 60.0f, L.field.getY() + 30.0f, L.field.getWidth() - 120.0f, L.field.getHeight() - 160.0f };
}

juce::Rectangle<float> Workstation::sliderBox (int k) const
{
    const auto p = planeRect();
    return { L.field.getX() + 110.0f, p.getBottom() + 22.0f + k * 24.0f, L.field.getWidth() - 220.0f, 14.0f };
}

Vec3 Workstation::cursorPoint() const
{
    return { juce::jlimit (-0.5, 0.5, (cursorRoot - kRootLo) / (kRootHi - kRootLo) - 0.5), juce::jlimit (-0.5, 0.5, cursorVoicing / kVoicingHi - 0.5), juce::jlimit (0.0, 1.0, cursorZ) };
}

Words Workstation::planeWords() const
{
    const auto near = st.nearest (cursorPoint(), 4, open);
    std::vector<std::pair<Chord, double>> parents;
    for (const auto& n : near)
        parents.push_back ({ n.item.stub >= 0 ? decompile (st.stubs[(size_t) n.item.stub].words, kDatumHz) : st.nodes[(size_t) n.item.node].chord, n.weight });
    if (parents.empty()) return Words {};
    return blendChords (parents);
}

void Workstation::setSlider (int k, juce::Point<float> p)
{
    const auto b = sliderBox (k);
    const double u = juce::jlimit (0.0, 1.0, (double) (p.x - b.getX()) / b.getWidth());
    if (k == 0) cursorRoot = kRootLo + u * (kRootHi - kRootLo);
    else if (k == 1) cursorVoicing = u * kVoicingHi;
    else cursorZ = u;
    playFrame = 0;
    playBody = false;
    juce::String names;
    for (const auto& n : st.nearest (cursorPoint(), 4, open)) { if (n.weight < 0.05 && names.isNotEmpty()) break; names += (names.isEmpty() ? "" : "   ") + juce::String ((int) std::round (n.weight * 100)) + "% " + st.itemName (n.item); }
    status = names;
    redraw();
}

void Workstation::takeCursor()
{
    int slot = selectedSlot;
    if (slot < 0) { for (int k = 0; k < 4; ++k) if (! chosen[(size_t) k]) { slot = k; break; } }
    if (slot < 0) slot = 3;
    const auto near = st.nearest (cursorPoint(), 1, open);
    int frame = -1;
    if (! near.empty() && near[0].weight > 0.999)
    {
        const auto& it = near[0].item;
        if (it.stub >= 0) frame = st.stubs[(size_t) it.stub].frame;
        else frame = st.nodes[(size_t) it.node].frame;
    }
    if (frame < 0) frame = lib.addNamed (planeWords(), "take " + noteName (cursorRoot) + " " + juce::String (cursorVoicing, 1) + "oct " + juce::String (widthFromZ (cursorZ), 2) + "st", kGroups - 1, true);
    body.corner[(size_t) slot] = frame;
    chosen[(size_t) slot] = true;
    fillCorners();
    editCorner = slot;
    selectedSlot = -1;
    status = juce::String (kCornerNames[slot]) + "  " + lib.frames[(size_t) frame].name;
}

void Workstation::fillCorners()
{
    const bool none = ! (chosen[0] || chosen[1] || chosen[2] || chosen[3]);
    if (none) for (int k = 0; k < 4; ++k) if (body.corner[(size_t) k] >= 0) chosen[(size_t) k] = true;
    if (! chosen[0])
        for (int k = 1; k < 4; ++k) if (chosen[(size_t) k]) { body.corner[0] = body.corner[(size_t) k]; break; }
    if (body.corner[0] < 0) return;
    if (! chosen[1]) body.corner[1] = body.corner[0];
    if (! chosen[2]) body.corner[2] = body.corner[0];
    if (! chosen[3]) body.corner[3] = body.corner[1];
}

void Workstation::paintPlane (Canvas& g)
{
    const auto p = planeRect();
    const float cw = p.getWidth() / kCols, ch = p.getHeight() / kRowsPlane;
    std::vector<int> count ((size_t) (kCols * kRowsPlane), 0);
    std::vector<double> res ((size_t) (kCols * kRowsPlane), 0.0);
    int most = 1;
    for (const auto& it : st.items)
    {
        if (! floorOpen (it.group)) continue;
        const int cx = juce::jlimit (0, kCols - 1, (int) ((it.root - kRootLo) / (kRootHi - kRootLo) * kCols));
        const int cy = juce::jlimit (0, kRowsPlane - 1, (int) (it.voicing / kVoicingHi * kRowsPlane));
        auto& c = count[(size_t) (cy * kCols + cx)];
        res[(size_t) (cy * kCols + cx)] += it.resonance;
        ++c;
        most = std::max (most, c);
    }
    for (int y = 0; y < kRowsPlane; ++y)
        for (int x = 0; x < kCols; ++x)
        {
            const int c = count[(size_t) (y * kCols + x)];
            if (c == 0) continue;
            const float a = (float) (std::log (1.0 + c) / std::log (1.0 + most));
            g.setColour (kData.withAlpha (0.15f + 0.85f * a));
            g.fillRect (juce::Rectangle<float> (p.getX() + x * cw + 1.0f, p.getBottom() - (y + 1) * ch + 1.0f, cw - 2.0f, ch - 2.0f));
        }
    g.setColour (kFrame);
    g.drawRect (px (p), 1);
    g.setColour (kDim);
    for (double note : { 36.0, 48.0, 60.0, 72.0, 84.0, 96.0, 108.0 })
    {
        const float x = p.getX() + (float) ((note - kRootLo) / (kRootHi - kRootLo)) * p.getWidth();
        g.drawText (noteName (note), (int) x - 14, (int) p.getBottom() + 4, 28, 12, juce::Justification::centred);
    }
    for (double oct : { 0.0, 2.0, 4.0, 6.0 })
    {
        const float y = p.getBottom() - (float) (oct / kVoicingHi) * p.getHeight();
        g.drawText (juce::String ((int) oct) + " oct", (int) p.getX() - 46, (int) y - 6, 40, 12, juce::Justification::centredRight);
    }
    const auto c = cursorPoint();
    const float cx = p.getX() + (float) (c.x + 0.5) * p.getWidth(), cy = p.getBottom() - (float) (c.y + 0.5) * p.getHeight();
    g.setColour (kLive);
    g.drawRect (juce::Rectangle<int> ((int) cx - 6, (int) cy - 6, 12, 12), 2);
    g.drawVerticalLine ((int) cx, p.getY(), p.getBottom());
    g.drawHorizontalLine ((int) cy, p.getX(), p.getRight());
    for (int k = 0; k < 4; ++k)
    {
        if (body.corner[(size_t) k] < 0) continue;
        const auto sh = Stitch::shapeOf (lib.frames[(size_t) body.corner[(size_t) k]].chord);
        const float kx = p.getX() + (float) ((sh.root - kRootLo) / (kRootHi - kRootLo)) * p.getWidth(), ky = p.getBottom() - (float) (sh.voicing / kVoicingHi) * p.getHeight();
        g.setColour (kChosen);
        g.drawRect (juce::Rectangle<int> ((int) kx - 5, (int) ky - 5, 10, 10), 1);
        g.drawText (kCornerNames[k], (int) kx + 8, (int) ky - 6, 40, 12, juce::Justification::centredLeft);
    }
    for (int k = 0; k < 3; ++k)
    {
        const auto b = sliderBox (k);
        const double u = k == 0 ? (cursorRoot - kRootLo) / (kRootHi - kRootLo) : k == 1 ? cursorVoicing / kVoicingHi : cursorZ;
        g.setColour (kDim);
        g.drawText (kAxisLabel[k], (int) L.field.getX() + 8, (int) b.getY() + 1, 96, 12, juce::Justification::centredLeft);
        g.setColour (kPanel);
        g.fillRect (b);
        g.setColour (kKeyLine);
        g.drawRect (px (b), 1);
        g.setColour (kLine);
        g.drawHorizontalLine ((int) b.getCentreY(), b.getX() + 2.0f, b.getRight() - 2.0f);
        const float kx = b.getX() + (float) u * b.getWidth();
        g.setColour (kLive);
        g.fillRect (juce::Rectangle<float> (kx - 3.0f, b.getY() + 1.0f, 6.0f, b.getHeight() - 2.0f));
        g.setColour (kText);
        const juce::String value = k == 0 ? noteName (cursorRoot) : k == 1 ? juce::String (cursorVoicing, 2) + " oct" : juce::String (widthFromZ (cursorZ), 2) + " st";
        g.drawText (value, (int) b.getRight() + 8, (int) b.getY() + 1, 96, 12, juce::Justification::centredLeft);
    }
    g.setColour (kText);
    g.drawText (status, (int) L.field.getX() + 8, (int) L.field.getY() + 8, (int) L.field.getWidth() - 16, 14, juce::Justification::centredLeft);
}

juce::Rectangle<float> Workstation::padRect() const
{
    const auto f = L.field;
    const float side = std::floor (std::min (f.getWidth(), f.getHeight()) - 120.0f);
    return { std::floor (f.getCentreX() - side * 0.5f), std::floor (f.getCentreY() - side * 0.5f + 10.0f), side, side };
}

void Workstation::setPad (juce::Point<float> p)
{
    const auto r = padRect();
    double m = (p.x - r.getX()) / r.getWidth(), q = (r.getBottom() - p.y) / r.getHeight();
    if (fine)
    {
        m = padPressM + (m - padPressM0) * 0.1;
        q = padPressQ + (q - padPressQ0) * 0.1;
    }
    body.morph = juce::jlimit (0.0, 1.0, m);
    body.q = juce::jlimit (0.0, 1.0, q);
    playBody = true;
    compare = false;
    redraw();
}

void Workstation::paintPad (Canvas& g)
{
    const auto r = padRect();
    g.setColour (kFrame);
    g.drawRect (px (r), 1);
    g.setColour (kLine);
    for (int i = 1; i < 4; ++i)
    {
        g.drawVerticalLine ((int) (r.getX() + r.getWidth() * i / 4.0f), r.getY(), r.getBottom());
        g.drawHorizontalLine ((int) (r.getY() + r.getHeight() * i / 4.0f), r.getX(), r.getRight());
    }
    for (int i = 0; i < 4; ++i)
    {
        const bool right = (i & 1) != 0, top = (i & 2) != 0;
        const int x = right ? (int) r.getRight() - 300 : (int) r.getX() + 60, y = top ? (int) r.getY() - 34 : (int) r.getBottom() + 24;
        const auto name = body.corner[(size_t) i] >= 0 ? lib.frames[(size_t) body.corner[(size_t) i]].name : juce::String ("empty");
        g.setColour (body.corner[(size_t) i] >= 0 ? kText : kDim);
        g.drawText (name, x, y, 240, 12, right ? juce::Justification::centredRight : juce::Justification::centredLeft);
    }
    g.setColour (compare ? kChosen : kDim);
    g.drawText (compare ? "comparing with " + juce::String (kCornerNames[0]) : "MORPH " + juce::String (body.morph, 3) + "   Q " + juce::String (body.q, 3), (int) r.getX(), (int) r.getBottom() + 44, (int) r.getWidth(), 12, juce::Justification::centred);
}
